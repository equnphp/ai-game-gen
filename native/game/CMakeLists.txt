# 游戏Runtime独立入口；禁止把作者Core/SQLite/Agent打入WASM。
cmake_minimum_required(VERSION 3.31.12)
project(GameGenGameRuntime VERSION 1.0.0 LANGUAGES C CXX)
if(NOT EMSCRIPTEN OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use the pinned Emscripten wasm32 toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GG_RUNTIME_ONLY ON)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/wasm-toolchain-profile.2.json" GG_WASM_PROFILE)
string(JSON GG_EMSCRIPTEN_VERSION GET "${GG_WASM_PROFILE}" emscripten version)
string(JSON GG_EMSCRIPTEN_COMMIT GET "${GG_WASM_PROFILE}" emscripten sourceCommit)
execute_process(COMMAND "${CMAKE_C_COMPILER}" --version OUTPUT_VARIABLE GG_EMCC_REPORT RESULT_VARIABLE GG_EMCC_STATUS)
if(NOT GG_EMCC_STATUS EQUAL 0 OR NOT GG_EMCC_REPORT MATCHES "${GG_EMSCRIPTEN_VERSION}" OR NOT GG_EMCC_REPORT MATCHES "${GG_EMSCRIPTEN_COMMIT}")
  message(FATAL_ERROR "Emscripten does not match the frozen profile")
endif()
string(JSON GG_COMPILE_COUNT LENGTH "${GG_WASM_PROFILE}" compileFlags)
math(EXPR GG_COMPILE_LAST "${GG_COMPILE_COUNT}-1")
foreach(GG_INDEX RANGE 0 ${GG_COMPILE_LAST})
  string(JSON GG_FLAG GET "${GG_WASM_PROFILE}" compileFlags ${GG_INDEX})
  if(GG_FLAG STREQUAL "-frtti" OR GG_FLAG STREQUAL "-std=c++20")
    add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:${GG_FLAG}>")
  else()
    add_compile_options("${GG_FLAG}")
  endif()
endforeach()
set(GG_RUNTIME_VARIANT development CACHE STRING "development or release")
if(NOT GG_RUNTIME_VARIANT STREQUAL "development" AND NOT GG_RUNTIME_VARIANT STREQUAL "release")
  message(FATAL_ERROR "Unknown runtime variant")
endif()
string(JSON GG_VARIANT_COUNT LENGTH "${GG_WASM_PROFILE}" variants ${GG_RUNTIME_VARIANT})
math(EXPR GG_VARIANT_LAST "${GG_VARIANT_COUNT}-1")
foreach(GG_INDEX RANGE 0 ${GG_VARIANT_LAST})
  string(JSON GG_FLAG GET "${GG_WASM_PROFILE}" variants ${GG_RUNTIME_VARIANT} ${GG_INDEX})
  if(GG_FLAG MATCHES "^-O" OR GG_FLAG MATCHES "^-g")
    add_compile_options("${GG_FLAG}")
  endif()
  add_link_options("${GG_FLAG}")
endforeach()
# 完整link参数将在ABI目标使用，静态运行库也采用同一异常/浮点/特性配置。
add_link_options(-fexceptions)
set(GG_BUILD_NODE "" CACHE FILEPATH "Explicit managed Node executable for build metadata")
if(NOT EXISTS "${GG_BUILD_NODE}")
  message(FATAL_ERROR "Set GG_BUILD_NODE to the managed Node executable")
endif()
set(GG_TOOLCHAIN_LOCK "${CMAKE_CURRENT_SOURCE_DIR}/toolchain-lock.windows.json" CACHE FILEPATH "Actual acquisition inventory")
get_filename_component(GG_EMSCRIPTEN_BIN "${CMAKE_C_COMPILER}" DIRECTORY)
set(GG_TOOL_SUFFIX "")
if(CMAKE_HOST_WIN32)
  set(GG_TOOL_SUFFIX ".exe")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/emcc-report.txt" "${GG_EMCC_REPORT}")
execute_process(COMMAND "${GG_EMSCRIPTEN_BIN}/../bin/clang${GG_TOOL_SUFFIX}" --version OUTPUT_VARIABLE GG_CLANG_REPORT RESULT_VARIABLE GG_CLANG_STATUS)
execute_process(COMMAND "${GG_EMSCRIPTEN_BIN}/../bin/wasm-opt${GG_TOOL_SUFFIX}" --version OUTPUT_VARIABLE GG_WASM_OPT_REPORT RESULT_VARIABLE GG_WASM_OPT_STATUS)
if(NOT GG_CLANG_STATUS EQUAL 0 OR NOT GG_WASM_OPT_STATUS EQUAL 0)
  message(FATAL_ERROR "Pinned LLVM/Binaryen reports unavailable")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/clang-report.txt" "${GG_CLANG_REPORT}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/wasm_opt-report.txt" "${GG_WASM_OPT_REPORT}")
execute_process(COMMAND "${GG_BUILD_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/build-profile.cjs" "${CMAKE_CURRENT_SOURCE_DIR}/.."
  "${CMAKE_CURRENT_SOURCE_DIR}/wasm-toolchain-profile.2.json" "${GG_TOOLCHAIN_LOCK}" "${GG_RUNTIME_VARIANT}"
  "${CMAKE_CURRENT_BINARY_DIR}/generated/runtime-inventory.json" RESULT_VARIABLE GG_PROFILE_STATUS)
if(NOT GG_PROFILE_STATUS EQUAL 0)
  message(FATAL_ERROR "Runtime input inventory failed")
endif()
file(READ "${CMAKE_CURRENT_BINARY_DIR}/generated/runtime-inventory.json" GG_INVENTORY)
string(JSON GG_GAME_PROFILE_ID GET "${GG_INVENTORY}" profileId)
string(JSON GG_GAME_PROFILE_HASH GET "${GG_INVENTORY}" profileHash)
string(REPLACE "sha256:" "" GG_GAME_PROFILE_HASH "${GG_GAME_PROFILE_HASH}")
string(JSON GG_GAME_BUILD_ID GET "${GG_INVENTORY}" runtimeBuild)
set(GG_GAME_DEVELOPMENT false)
if(GG_RUNTIME_VARIANT STREQUAL "development")
  set(GG_GAME_DEVELOPMENT true)
endif()
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/game-build.hpp.in" generated/game-build.hpp @ONLY)
# 源码改变必须同时更新嵌入身份；不能编译新对象却沿用旧profileHash。
string(JSON GG_INPUT_COUNT LENGTH "${GG_INVENTORY}" identity inputs)
math(EXPR GG_INPUT_LAST "${GG_INPUT_COUNT}-1")
foreach(GG_INDEX RANGE 0 ${GG_INPUT_LAST})
  string(JSON GG_INPUT_NAME MEMBER "${GG_INVENTORY}" identity inputs ${GG_INDEX})
  get_filename_component(GG_INPUT_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../${GG_INPUT_NAME}" ABSOLUTE)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${GG_INPUT_PATH}")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${GG_TOOLCHAIN_LOCK}" "${CMAKE_CURRENT_SOURCE_DIR}/wasm-toolchain-profile.2.json")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../engine/node-catalog.json" GG_NODE_CATALOG)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../engine/ir1_2.schema.json" GG_IR_SCHEMA)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../engine/runtime_bundle.schema.json" GG_BUNDLE_SCHEMA)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../engine/runtime_resources.schema.json" GG_RUNTIME_RESOURCES_SCHEMA)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../engine/runtime_port_1_0.schema.json" GG_RUNTIME_PORT_1_0)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../engine/runtime_port_1_1.schema.json" GG_RUNTIME_PORT_1_1)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/runtime-contracts.hpp.in" generated/contracts.hpp @ONLY)
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/Dependencies.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/RuntimeLibraries.cmake")
add_executable(gg_game abi.cpp identity.cpp)
target_include_directories(gg_game PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(gg_game PRIVATE gg_runtime)
set_target_properties(gg_game PROPERTIES OUTPUT_NAME "game-runtime.generated" SUFFIX ".js")
string(JSON GG_LINK_COUNT LENGTH "${GG_WASM_PROFILE}" linkFlags)
math(EXPR GG_LINK_LAST "${GG_LINK_COUNT}-1")
foreach(GG_INDEX RANGE 0 ${GG_LINK_LAST})
  string(JSON GG_FLAG GET "${GG_WASM_PROFILE}" linkFlags ${GG_INDEX})
  if(GG_FLAG MATCHES "^-sEXPORTED_FUNCTIONS=@")
    set(GG_FLAG "-sEXPORTED_FUNCTIONS=@${CMAKE_CURRENT_SOURCE_DIR}/game-exports-1.1.json")
  endif()
  target_link_options(gg_game PRIVATE "${GG_FLAG}")
endforeach()
target_link_options(gg_game PRIVATE "--js-library=${CMAKE_CURRENT_SOURCE_DIR}/host-library.js")
add_custom_command(TARGET gg_game POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E copy "$<TARGET_FILE_DIR:gg_game>/game-runtime.generated.wasm" "${CMAKE_CURRENT_BINARY_DIR}/game-runtime.wasm"
  COMMAND "${GG_BUILD_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/wrap-runtime.cjs" "${CMAKE_CURRENT_SOURCE_DIR}/factory-wrapper.cjs"
    "$<TARGET_FILE:gg_game>" "${CMAKE_CURRENT_BINARY_DIR}/game-runtime.js" "${CMAKE_CURRENT_BINARY_DIR}/generated/runtime-inventory.json"
  VERBATIM)
