# 游戏Runtime独立入口；禁止把作者Core/SQLite/Agent打入WASM。
cmake_minimum_required(VERSION 3.31.12)
project(GameGenGameRuntime VERSION 1.0.0 LANGUAGES C CXX)
if(NOT EMSCRIPTEN OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use the pinned Emscripten wasm32 toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GG_RUNTIME_ONLY ON)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/wasm-toolchain-profile.json" GG_WASM_PROFILE)
string(JSON GG_EMSCRIPTEN_VERSION GET "${GG_WASM_PROFILE}" emscripten version)
string(JSON GG_EMSCRIPTEN_COMMIT GET "${GG_WASM_PROFILE}" emscripten sourceCommit)
execute_process(COMMAND "${CMAKE_C_COMPILER}" --version OUTPUT_VARIABLE GG_EMCC_REPORT RESULT_VARIABLE GG_EMCC_STATUS)
if(NOT GG_EMCC_STATUS EQUAL 0 OR NOT GG_EMCC_REPORT MATCHES "${GG_EMSCRIPTEN_VERSION}" OR NOT GG_EMCC_REPORT MATCHES "${GG_EMSCRIPTEN_COMMIT}")
  message(FATAL_ERROR "Emscripten does not match the frozen profile")
endif()
string(JSON GG_COMPILE_COUNT LENGTH "${GG_WASM_PROFILE}" compileFlags)
math(EXPR GG_COMPILE_LAST "${GG_COMPILE_COUNT}-1")
foreach(GG_INDEX RANGE 0 ${GG_COMPILE_LAST})
  string(JSON GG_FLAG GET "${GG_WASM_PROFILE}" compileFlags ${GG_INDEX})
  if(GG_FLAG STREQUAL "-frtti" OR GG_FLAG STREQUAL "-std=c++20")
    add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:${GG_FLAG}>")
  else()
    add_compile_options("${GG_FLAG}")
  endif()
endforeach()
set(GG_RUNTIME_VARIANT development CACHE STRING "development or release")
if(NOT GG_RUNTIME_VARIANT STREQUAL "development" AND NOT GG_RUNTIME_VARIANT STREQUAL "release")
  message(FATAL_ERROR "Unknown runtime variant")
endif()
string(JSON GG_VARIANT_COUNT LENGTH "${GG_WASM_PROFILE}" variants ${GG_RUNTIME_VARIANT})
math(EXPR GG_VARIANT_LAST "${GG_VARIANT_COUNT}-1")
foreach(GG_INDEX RANGE 0 ${GG_VARIANT_LAST})
  string(JSON GG_FLAG GET "${GG_WASM_PROFILE}" variants ${GG_RUNTIME_VARIANT} ${GG_INDEX})
  if(GG_FLAG MATCHES "^-O" OR GG_FLAG MATCHES "^-g")
    add_compile_options("${GG_FLAG}")
  endif()
  add_link_options("${GG_FLAG}")
endforeach()
# 完整link参数将在ABI目标使用，静态运行库也采用同一异常/浮点/特性配置。
add_link_options(-fexceptions)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../engine/node-catalog.json" GG_NODE_CATALOG)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../engine/ir1_2.schema.json" GG_IR_SCHEMA)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../engine/runtime_bundle.schema.json" GG_BUNDLE_SCHEMA)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/runtime-contracts.hpp.in" generated/contracts.hpp @ONLY)
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/Dependencies.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/RuntimeLibraries.cmake")
